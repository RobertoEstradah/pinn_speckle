% ====== Abstract en inglés ======
\begin{center}
  \textbf{Abstract}
\end{center}
\begin{quote}
\noindent\singlespacing
Physics-Informed Neural Networks with sinusoidal activation functions (SIREN-PINNs)
are applied to simulate optical speckle patterns by solving the 2D Helmholtz equation
with a stochastic phase boundary condition.
The proposed SIREN architecture ($5 \times 128$, $\omega_0 = 1.0$) achieves relative
$L^2$ errors of $0.006\%$ (1D) and $0.171\%$ (2D) against analytical solutions,
outperforming the state-of-the-art by factors of $\times 415$ and $\times 11.2$,
respectively.
Speckle patterns are generated by imposing a random phase condition
$\phi(x) \sim \mathcal{U}(0, 2\pi)$ on the illuminated surface and solving the
Helmholtz equation freely in the interior domain.
Statistical validation follows Goodman's theory: the speckle contrast
$C = \sigma_I / \langle I \rangle \approx 1$ and intensity distributions consistent
with the negative exponential confirm fully developed speckle.
A two-phase Adam + L-BFGS optimization strategy and Latin Hypercube Sampling (LHS)
are key to convergence.
Results establish SIREN-PINNs as a computationally efficient and physically accurate
surrogate for optical speckle simulation.
\end{quote}

\vspace{0.8em}
\noindent \textbf{Keywords:} Physics-Informed Neural Networks, SIREN,
Helmholtz equation, optical speckle, sinusoidal activation, Latin Hypercube Sampling.

\vspace{1.2em}

% ====== Resumen en español ======
\begin{center}
  \textbf{Resumen}
\end{center}
\begin{abstract}
\noindent\singlespacing
Se presenta una implementación de Redes Neuronales Informadas por Física con
activación sinusoidal (PINN-SIREN) para la simulación acelerada del patrón de
speckle óptico, mediante la resolución de la ecuación de Helmholtz bidimensional
con campo eléctrico complejo y condición de frontera de fase aleatoria.
La arquitectura SIREN propuesta ($5 \times 128$ neuronas, $\omega_0 = 1.0$)
alcanza errores $L^2$ de $0.006\%$ en el caso 1D y $0.171\%$ en el caso 2D
respecto a soluciones analíticas, superando al estado del arte en factores de
$\times 415$ y $\times 11.2$, respectivamente.
El patrón de speckle se genera imponiendo una fase aleatoria
$\phi(x) \sim \mathcal{U}(0, 2\pi)$ en la superficie iluminada y resolviendo la
ecuación de Helmholtz libremente en el interior del dominio.
La validación estadística sigue el criterio de Goodman: contraste
$C = \sigma_I / \langle I \rangle \approx 1$ y distribución de intensidades
exponencial negativa, confirmando speckle completamente desarrollado.
\end{abstract}

\vspace{0.5em}
\noindent \textbf{Palabras clave:} Redes Neuronales Informadas por Física,
SIREN, ecuación de Helmholtz, speckle óptico, activación sinusoidal,
muestreo por hipercubo latino.
